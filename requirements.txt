# Librerías que necesita el proyecto
# Se instalan todas con este comando: pip install -r requirements.txt

# FastAPI es el framework con el que se hace la API
fastapi
# uvicorn es el servidor que pone a correr la API
uvicorn
# SQLAlchemy sirve para trabajar con la base de datos usando clases de Python
sqlalchemy
# psycopg es el conector entre Python y PostgreSQL
psycopg[binary]
# pydantic valida los datos que llegan (el [email] es para poder validar correos)
pydantic[email]
# python-dotenv sirve para leer el archivo .env
python-dotenv
# PyJWT sirve para crear y leer los tokens JWT
PyJWT
# bcrypt sirve para encriptar las contraseñas
bcrypt
# pytest sirve para hacer las pruebas automáticas
pytest
# httpx lo necesita el TestClient que se usa en las pruebas
httpx
